```python
"""Utilidades comuns do Life: fuso fixo, abertura só leitura, itens e interface dos provedores."""
from __future__ import annotations

import os
import sqlite3
from contextlib import contextmanager
from datetime import date, datetime, timedelta, timezone
from pathlib import Path
from typing import Iterable, Iterator, Optional

LOCAL = timezone(timedelta(hours=-3))  # UTC-3 fixo, igual ao nas-lifelog-export (DST até 2019 e viagens: limitação conhecida)


def find_db(env: str, candidates: Iterable[str]) -> Optional[Path]:
    v = os.environ.get(env, "") if env else ""
    for c in ([v] if v else []) + list(candidates):
        if c and Path(c).is_file():
            return Path(c)
    return None


@contextmanager
def ro(path: Path) -> Iterator[sqlite3.Connection]:
    con = sqlite3.connect(f"file:{path}?mode=ro", uri=True, timeout=10)
    con.row_factory = sqlite3.Row
    try:
        yield con
    finally:
        con.close()


def parse_ts(value) -> Optional[datetime]:
    """ISO 8601 ('Z', '+00:00' ou sem fuso = UTC) -> datetime aware em UTC."""
    if not value:
        return None
    try:
        dt = datetime.fromisoformat(str(value).strip().replace("Z", "+00:00"))
    except ValueError:
        return None
    if dt.tzinfo is None:
        dt = dt.replace(tzinfo=timezone.utc)
    return dt.astimezone(timezone.utc)


def local_day(dt: datetime) -> str:
    return dt.astimezone(LOCAL).date().isoformat()


def hm(dt: Optional[datetime]) -> Optional[str]:
    return dt.astimezone(LOCAL).strftime("%H:%M") if dt else None


def today() -> date:
    return datetime.now(LOCAL).date()


def make_item(*, id: str, kind: str, source: str, title: str, ts: Optional[datetime] = None,
              end_ts: Optional[datetime] = None, subtitle: Optional[str] = None, url: Optional[str] = None,
              open_url: Optional[str] = None, day: Optional[str] = None, **meta) -> dict:
    return {
        "id": id, "kind": kind, "source": source,
        "ts": ts.astimezone(LOCAL).isoformat() if ts else None,
        "end_ts": end_ts.astimezone(LOCAL).isoformat() if end_ts else None,
        "time": hm(ts), "end_time": hm(end_ts),
        "day": day or (local_day(ts) if ts else None),
        "title": title, "subtitle": subtitle, "url": url,
        "open_url": open_url if open_url is not None else url,
        "place_ref": None, "meta": meta,
    }


def status(name: str, label: str, state: str, **extra) -> dict:
    return {"name": name, "label": label, "state": state, **extra}


class Provider:
    """Interface das fontes do Life. O arquivo é resolvido a cada chamada (env > candidatos)."""
    name = ""
    label = ""
    env = ""
    candidates: tuple = ()

    def path(self) -> Optional[Path]:
        return find_db(self.env, self.candidates)

    def status(self) -> dict:
        raise NotImplementedError

    def day(self, d: date) -> list[dict]:
        return []

    def stats(self, d: date) -> dict:
        return {}

    def counts(self, a: date, b: date) -> dict:
        return {}

    def search(self, q: str, limit: int) -> list[dict]:
        return []


def safe_status(p: Provider) -> dict:
    try:
        return p.status()
    except Exception as e:  # banco corrompido/trancado: a fonte aparece com erro, o resto segue
        return status(p.name, p.label, "error", detail=str(e)[:200])
```

```python
"""Formatação dos textos do Life (UI em inglês)."""
from __future__ import annotations

SYMBOLS = {"BRL": "R$", "USD": "US$", "EUR": "€", "GBP": "£"}


def money(cents: int, currency: str) -> str:
    v = abs(cents) / 100
    sym = SYMBOLS.get(currency, currency)
    return f"{sym} {v:,.0f}" if v >= 100 else f"{sym} {v:,.2f}"


def compact(n) -> str:
    n = float(n)
    for div, suf in ((1e6, "M"), (1e3, "k")):
        if abs(n) >= div:
            return f"{n / div:.1f}".rstrip("0").rstrip(".") + suf
    return f"{n:.0f}"


def plural(n: int, word: str) -> str:
    return f"{n} {word}{'' if n == 1 else 's'}"
```

```python
"""Limpeza dos títulos do Google Takeout (Minha atividade) para a linha do tempo do Life.

O exportador grava o título cru ("Watched …", "Searched for …", "Visited https://…"). Aqui o verbo vira `action`,
o resto vira o título, e cada evento cai num grupo da linha "Digital" (src/life/assemble.py)."""
from __future__ import annotations

import re
from typing import Optional
from urllib.parse import urlparse

VERBS = (  # prefixos mais longos primeiro
    ("Searched for ", "searched"), ("Directions to ", "directions"), ("Watched ", "watched"),
    ("Visited ", "visited"), ("Viewed ", "viewed"), ("Used ", "used"), ("Said ", "said"),
    ("Explored ", "explored"), ("Liked ", "liked"), ("Launched ", "launched"), ("Received ", "received"),
    ("Read ", "read"), ("Saw ", "saw"), ("Opened ", "opened"), ("Translated ", "translated"),
    ("Searched ", "searched"),
)
# Telemetria/anúncios: fora do dia e das contagens; a busca ainda acha.
NOISE_SOURCES = frozenset({"other:Google Analytics", "other:Ads", "other:My Ad Center", "other:Help",
                           "other:Voice Match", "other:Takeout"})
GROUP_LABELS = {  # grupo -> (singular, plural, título da sublista)
    "search": ("search", "searches", "Searches"),
    "web": ("page", "pages", "Pages"),
    "video": ("video", "videos", "YouTube"),
    "news": ("article", "articles", "News"),
    "maps": ("Maps action", "Maps actions", "Maps"),
    "apps": ("app", "apps", "Play Store"),
    "assistant": ("Assistant request", "Assistant requests", "Assistant"),
    "tv": ("TV item", "TV items", "Google TV"),
    "other": ("other item", "other items", "Other"),
}
GROUP_ORDER = tuple(GROUP_LABELS)
_URL = re.compile(r"^https?://", re.I)


def domain(url: Optional[str]) -> Optional[str]:
    if not url:
        return None
    try:
        host = urlparse(url).hostname or ""
    except ValueError:
        return None
    return (host[4:] if host.startswith("www.") else host) or None


def _group(source: str, action: str) -> str:
    if source == "youtube_watch":
        return "search" if action == "searched" else "video"
    if source == "search":
        return "search" if action == "searched" else "web"
    return {"chrome": "web", "maps": "maps", "play": "apps", "assistant": "assistant", "other:Google TV": "tv",
            "other:Google News": "news", "other:Discover": "news"}.get(source, "other")


def clean_event(source: str, title: Optional[str], url: Optional[str], detail: Optional[str]) -> dict:
    raw = (title or "").strip()
    action, text = "", raw
    for prefix, verb in VERBS:
        if raw.startswith(prefix):
            action, text = verb, raw[len(prefix):].strip()
            break
    group = _group(source, action)
    unavailable = False
    subtitle = detail.strip() if detail and detail.strip() else None
    if group == "video" and (not text or _URL.match(text)):
        text, unavailable = "YouTube video (unavailable)", True
    elif _URL.match(text):  # "Visited https://…" sem título de página: domínio + caminho
        path = urlparse(text).path
        text = (domain(text) or text) + (path if path not in ("", "/") else "")
    if not subtitle and group in ("web", "news") and url:
        subtitle = domain(url)
    if source.startswith("other:") and group == "other":
        subtitle = subtitle or source[6:]
    base = (text or raw).casefold()
    key = f"{group}|{action}|{base}" + (f"|{url}" if group in ("video", "web") and url else "")
    return {"group": group, "action": action or "used", "title": text or raw or "(untitled)", "subtitle": subtitle,
            "unavailable": unavailable, "noise": source in NOISE_SOURCES, "key": key}


def count_phrase(group: str, n: int) -> str:
    one, many, _ = GROUP_LABELS[group]
    return f"{n} {one if n == 1 else many}"
```

```python
"""Provedor do Google Takeout para o Life: lê o lifelog.sqlite do nas-lifelog-export (só leitura).

Atividade digital (Minha atividade) vira `digital` com título limpo (src/life/clean.py); treinos do Fit viram `health`;
notas do Keep criadas/editadas no dia viram `note`. O Fit diário alimenta os números do dia."""
from __future__ import annotations

from datetime import date
from urllib.parse import quote_plus

from src.life.clean import NOISE_SOURCES, clean_event
from src.life.common import Provider, local_day, make_item, parse_ts, ro, status

GOOGLE_SEARCH = "https://www.google.com/search?q="
CONTEXT_HITS = 100  # acertos da busca que ganham vizinhos de contexto


def event_item(r, day: str) -> dict:
    c = clean_event(r["source"], r["title"], r["url"], r["detail"])
    open_url = r["url"]
    if not open_url and r["source"] == "search" and c["action"] == "searched":
        open_url = GOOGLE_SEARCH + quote_plus(c["title"])
    return make_item(id=f"tk:{r['id']}", kind="digital", source=r["source"], ts=parse_ts(r["ts"]), title=c["title"],
                     subtitle=c["subtitle"], url=r["url"], open_url=open_url, day=day, group=c["group"],
                     action=c["action"], unavailable=c["unavailable"], noise=c["noise"], key=c["key"])


def _note_title(r) -> str:
    t = (r["title"] or "").strip()
    if t:
        return t
    first = next((ln.strip() for ln in (r["text"] or "").splitlines() if ln.strip()), "")
    return first[:80] or "Untitled note"


class TakeoutProvider(Provider):
    name, label = "takeout", "Google Takeout"
    env = "LIFELOG_DB"
    candidates = ("/app/lifelog/lifelog.sqlite", "/data/lifelog/lifelog.sqlite")

    def status(self) -> dict:
        p = self.path()
        if not p:
            return status(self.name, self.label, "missing",
                          detail="run nas-lifelog-export on the NAS and lifelog-sync on the PC")
        with ro(p) as con:
            a, b = con.execute("select min(day), max(day) from events").fetchone()
            fa, fb = con.execute("select min(day), max(day) from fit_daily").fetchone()
            na, nb = con.execute("select min(substr(coalesce(edited, created),1,10)), "
                                 "max(substr(coalesce(edited, created),1,10)) from notes where trashed=0").fetchone()
            built = dict(con.execute("select key, value from meta").fetchall()).get("built_at")
        return status(self.name, self.label, "ok", built_at=built,
                      coverage={"digital": [a, b], "steps": [fa, fb], "notes": [na, nb]}, **{"from": a, "to": b})

    def day(self, d: date) -> list[dict]:
        p = self.path()
        if not p:
            return []
        ds, out = d.isoformat(), []
        with ro(p) as con:
            out += [event_item(r, ds) for r in con.execute(
                "select id, ts, source, title, url, detail from events where day=? order by ts", (ds,))]
            for r in con.execute("select start, end, activity, distance_m, duration_s from fit_sessions "
                                 "where substr(start,1,10) between date(?,'-1 day') and date(?,'+1 day') order by start",
                                 (ds, ds)):
                st = parse_ts(r["start"])
                if not st or local_day(st) != ds:
                    continue
                mins, km = round((r["duration_s"] or 0) / 60), (r["distance_m"] or 0) / 1000
                sub = " · ".join(x for x in (f"{mins} min" if mins else "", f"{km:.1f} km" if km >= 0.05 else "") if x)
                out.append(make_item(id=f"fit:{r['start']}:{r['activity']}", kind="health", source="fit", ts=st,
                                     end_ts=parse_ts(r["end"]),
                                     title=(r["activity"] or "workout").replace("_", " ").capitalize(),
                                     subtitle=sub or None, day=ds))
            for r in con.execute("select id, title, text, created, edited from notes where trashed=0 and ("
                                 "substr(created,1,10) between date(?,'-1 day') and date(?,'+1 day') or "
                                 "substr(edited,1,10) between date(?,'-1 day') and date(?,'+1 day'))", (ds,) * 4):
                cr, ed = parse_ts(r["created"]), parse_ts(r["edited"])
                if cr and local_day(cr) == ds:
                    when, verb = cr, "created"
                elif ed and local_day(ed) == ds:
                    when, verb = ed, "edited"
                else:
                    continue
                out.append(make_item(id=f"keep:{r['id']}", kind="note", source="keep", ts=when, title=_note_title(r),
                                     subtitle=f"Keep · {verb}", day=ds))
        return out

    def stats(self, d: date) -> dict:
        p = self.path()
        if not p:
            return {}
        with ro(p) as con:
            r = con.execute("select steps, distance_m, move_minutes, calories from fit_daily where day=?",
                            (d.isoformat(),)).fetchone()
        if not r:
            return {}
        vals = {"steps": r["steps"], "distance_km": round((r["distance_m"] or 0) / 1000, 1),
                "move_minutes": r["move_minutes"], "calories": round(r["calories"]) if r["calories"] else None}
        return {k: v for k, v in vals.items() if v}

    def counts(self, a: date, b: date) -> dict:
        p = self.path()
        if not p:
            return {}
        out: dict = {}
        noise = tuple(sorted(NOISE_SOURCES))
        span = (a.isoformat(), b.isoformat())
        with ro(p) as con:
            for day, n in con.execute(f"select day, count(*) from events where day between ? and ? and source not in "
                                      f"({','.join('?' * len(noise))}) group by day", (*span, *noise)):
                out.setdefault(day, {})["digital"] = n
            for day, steps in con.execute("select day, steps from fit_daily where day between ? and ? and steps > 0", span):
                out.setdefault(day, {})["steps"] = steps
            for day, n in con.execute("select substr(coalesce(edited, created),1,10) d, count(*) from notes "
                                      "where trashed=0 and d between ? and ? group by d", span):
                out.setdefault(day, {})["notes"] = n
        return out

    def search(self, q: str, limit: int) -> list[dict]:
        p = self.path()
        if not p:
            return []
        like, out = f"%{q}%", []
        with ro(p) as con:
            rows = con.execute("select id, ts, day, source, title, url, detail from events "
                               "where title like ? or detail like ? order by ts desc limit ?", (like, like, limit)).fetchall()
            for i, r in enumerate(rows):
                it = event_item(r, r["day"])
                if i < CONTEXT_HITS:
                    it["meta"]["context"] = self._context(con, r)
                out.append(it)
            for r in con.execute("select id, title, text, created, edited from notes where trashed=0 and "
                                 "(title like ? or text like ?) order by coalesce(edited, created) desc limit ?",
                                 (like, like, limit)):
                out.append(make_item(id=f"keep:{r['id']}", kind="note", source="keep",
                                     ts=parse_ts(r["edited"]) or parse_ts(r["created"]), title=_note_title(r),
                                     subtitle="Keep"))
        return out

    @staticmethod
    def _context(con, r) -> list[dict]:
        out = []
        for rel, op, order in (("before", "<", "desc"), ("after", ">", "asc")):
            n = con.execute(f"select id, ts, source, title, url, detail from events where day=? and ts {op} ? "
                            f"order by ts {order} limit 1", (r["day"], r["ts"])).fetchone()
            if n:
                it = event_item(n, r["day"])
                out.append({"rel": rel, "time": it["time"], "title": it["title"], "group": it["meta"]["group"]})
        return out
```

```python
def _cat_label(cat: str) -> str:
    last = (cat or "").split(":")[-1].replace("-", " ").replace("_", " ").strip()
    return last[:1].upper() + last[1:] if last else ""


def _neutral(cat: Optional[str]) -> bool:
    return any((cat or "").startswith(p) for p in NEUTRAL_PREFIXES)


class MoneyProvider(Provider):

        return find_db(self.env, ([f"{root}/finance.sqlite"] if root else []) + list(self.candidates))

    def status(self) -> dict:

                         day=r["date"], amount_cents=r["amount_cents"], currency=r["currency"], category=cat or None,

            if m["neutral"] or not m["amount_cents"]:
                continue
            bucket = spent if m["amount_cents"] < 0 else income
            bucket[m["currency"]] = bucket.get(m["currency"], 0) + abs(m["amount_cents"])
        return {k: v for k, v in (("spent", spent), ("income", income)) if v}

    def counts(self, a: date, b: date) -> dict:
        p = self.path()
        if not p:
            return {}
        out: dict = {}
        with ro(p) as con:
            for r in con.execute(SQL + "where l.date between ? and ? and a.currency='BRL' and l.amount_cents < 0",
                                 (a.isoformat(), b.isoformat())):
                if _neutral(r["category"]):
                    continue
                day = out.setdefault(r["date"], {"spent": 0})
                day["spent"] += -r["amount_cents"]
        return out

    def search(self, q: str, limit: int) -> list[dict]:
        p = self.path()
        if not p:
            return []
        with ro(p) as con:
            return [self._item(r) for r in con.execute(SQL + "where l.description like ? order by l.date desc limit ?",
                                                        (f"%{q}%", limit))]
```

```python
"""Monta a página do dia do Life no servidor; a UI só desenha.

Fase 1 (sem lugares): a espinha são os blocos do dia (Night 0–6, Morning 6–12, Afternoon 12–18, Evening 18–24),
só os que têm itens. Itens sem hora (finanças) vão para a faixa "All day". Corridas de itens digitais viram um
grupo "Digital · 6 searches, 2 videos (3 views)" — a corrida quebra num item não digital ou numa lacuna > 60 min."""
from __future__ import annotations

from collections import Counter
from datetime import date

from src.life.clean import GROUP_LABELS, GROUP_ORDER, count_phrase
from src.life.common import safe_status
from src.life.fmt import compact, money, plural

PERIODS = (("night", "Night", 0, 6), ("morning", "Morning", 6, 12), ("afternoon", "Afternoon", 12, 18),
           ("evening", "Evening", 18, 24))
RUN_GAP_MIN = 60


def _minutes(hhmm: str) -> int:
    h, m = hhmm.split(":")
    return int(h) * 60 + int(m)


def digital_group(items: list[dict]) -> dict:
    by_group: dict = {}
    for it in items:
        m = it["meta"]
        rows = by_group.setdefault(m["group"], {})
        row = rows.get(m["key"])
        if row:
            row["times"].append(it["time"])
            row["count"] += 1
        else:
            rows[m["key"]] = {"id": it["id"], "title": it["title"], "subtitle": it["subtitle"], "url": it["url"],
                              "open_url": it["open_url"], "action": m["action"], "unavailable": m["unavailable"],
                              "times": [it["time"]], "count": 1}
    sources, parts = [], []
    for g in GROUP_ORDER:
        if g not in by_group:
            continue
        rows = sorted(by_group[g].values(), key=lambda r: (r["unavailable"], r["times"][0]))
        n, views = len(rows), sum(r["count"] for r in rows)
        parts.append(count_phrase(g, n) + (f" ({views} views)" if g == "video" and views != n else ""))
        sources.append({"group": g, "label": GROUP_LABELS[g][2], "count": n, "views": views,
                        "unavailable": sum(1 for r in rows if r["unavailable"]), "items": rows})
    return {"kind": "digital_group", "id": "dg:" + items[0]["id"], "time": items[0]["time"],
            "end_time": items[-1]["time"], "total": sum(s["count"] for s in sources), "summary": ", ".join(parts),
            "unavailable": sum(s["unavailable"] for s in sources), "sources": sources}


def entries_for(items: list[dict]) -> list[dict]:
    out: list = []
    run: list = []
    for it in items:
        if it["kind"] == "digital":
            if run and _minutes(it["time"]) - _minutes(run[-1]["time"]) > RUN_GAP_MIN:
                out.append(digital_group(run))
                run = []
            run.append(it)
        else:
            if run:
                out.append(digital_group(run))
                run = []
            out.append(it)
    if run:
        out.append(digital_group(run))
    return out


def summary(counts: Counter, stats: dict) -> dict:
    parts = []
    if stats.get("spent"):
        parts.append({"kind": "money", "text": " + ".join(money(c, cur) for cur, c in stats["spent"].items()) + " spent"})
    if stats.get("steps"):
        parts.append({"kind": "health", "text": f"{compact(stats['steps'])} steps"})
    if counts.get("work"):
        parts.append({"kind": "work", "text": plural(counts["work"], "AI session")})
    if counts.get("note"):
        parts.append({"kind": "note", "text": plural(counts["note"], "note")})
    if counts.get("digital"):
        parts.append({"kind": "digital", "text": f"{counts['digital']} digital"})
    return {"text": " · ".join(p["text"] for p in parts), "parts": parts}


def stat_rows(stats: dict, counts: Counter) -> list[dict]:
    rows: list = []

    def add(key, label, value):
        if value:
            rows.append({"key": key, "label": label, "value": value})

    add("steps", "Steps", f"{stats['steps']:,}" if stats.get("steps") else None)
    add("distance", "Distance", f"{stats['distance_km']:.1f} km" if stats.get("distance_km") else None)
    add("move", "Active", f"{stats['move_minutes']} min" if stats.get("move_minutes") else None)
    for cur, c in (stats.get("spent") or {}).items():
        add("spent", "Spent", money(c, cur))
    for cur, c in (stats.get("income") or {}).items():
        add("income", "Received", money(c, cur))
    add("ai", "AI sessions", str(stats["ai_sessions"]) if stats.get("ai_sessions") else None)
    add("tokens", "AI tokens", compact(stats["ai_tokens"]) if stats.get("ai_tokens") else None)
    add("digital", "Digital items", str(counts["digital"]) if counts.get("digital") else None)
    return rows


def assemble_day(d: date, providers) -> dict:
    items: list = []
    stats: dict = {}
    sources: list = []
    for p in providers:
        st = safe_status(p)
        if st["state"] == "ok":
            try:
                items += p.day(d)
                stats.update(p.stats(d))
            except Exception as e:  # uma fonte quebrada não derruba o dia
                st = {**st, "state": "error", "detail": str(e)[:200]}
        sources.append(st)
    items = [i for i in items if not i["meta"].get("noise")]
    all_day = [i for i in items if not i["time"]]
    timed = sorted((i for i in items if i["time"]), key=lambda i: i["ts"])
    blocks = []
    for key, label, h0, h1 in PERIODS:
        its = [i for i in timed if h0 * 60 <= _minutes(i["time"]) < h1 * 60]
        if its:
            blocks.append({"kind": "period", "key": key, "label": label, "start": f"{h0:02d}:00",
                           "end": f"{h1:02d}:00", "entries": entries_for(its)})
    counts = Counter(i["kind"] for i in items)
    return {"day": d.isoformat(), "weekday": d.strftime("%A"), "summary": summary(counts, stats), "all_day": all_day,
            "blocks": blocks, "counts_by_kind": dict(counts), "stats": stat_rows(stats, counts), "trip": None,
            "sources": sources}
```

```python
"""Visões de resumo do Life: mês, ano, "On this day", busca e dias vizinhos com dados.

`values[métrica]` é None fora da cobertura da fonte (sem dado) e 0 dentro dela — o mapa de calor separa os dois.
Faixas fixas por métrica (nunca relativas a um dia extremo): ver docs/superpowers/research/…referencias-ui-life-log.md."""
from __future__ import annotations

from collections import Counter
from datetime import date, timedelta
from typing import Iterator, Optional

from src.life.common import safe_status
from src.life.fmt import compact, money, plural

MONTHS = ("January", "February", "March", "April", "May", "June", "July", "August", "September", "October",
          "November", "December")
METRICS = (  # chave, rótulo, limite inferior dos níveis 1..4, unidade
    ("digital", "Digital", (1, 20, 60, 120), "items"),
    ("spent", "Spent", (1, 5000, 20000, 50000), "cents"),
    ("steps", "Steps", (1, 3000, 7000, 12000), "steps"),
    ("work", "AI sessions", (1, 2, 4, 8), "sessions"),
    ("notes", "Notes", (1, 2, 4, 8), "notes"),
)
KEYS = tuple(m[0] for m in METRICS)


def metrics_meta() -> list[dict]:
    return [{"key": k, "label": label, "bands": list(b), "unit": u} for k, label, b, u in METRICS]


def _span(a: date, b: date) -> Iterator[date]:
    d = a
    while d <= b:
        yield d
        d += timedelta(days=1)


def _collect(providers, a: date, b: date):
    days: dict = {}
    coverage: dict = {}
    for p in providers:
        st = safe_status(p)
        if st["state"] != "ok":
            continue
        coverage.update(st.get("coverage") or {})
        try:
            got = p.counts(a, b)
        except Exception:
            continue
        for ds, vals in got.items():
            days.setdefault(ds, {}).update(vals)
    return days, coverage


def _values(ds: str, raw: dict, coverage: dict) -> dict:
    out = {}
    for k in KEYS:
        span = coverage.get(k)
        covered = bool(span and span[0] and span[1] and span[0] <= ds <= span[1])
        out[k] = raw.get(k, 0) if covered else raw.get(k)
    return out


def counts_summary(v: dict) -> str:
    parts = []
    if v.get("spent"):
        parts.append(money(v["spent"], "BRL") + " spent")
    if v.get("steps"):
        parts.append(compact(v["steps"]) + " steps")
    if v.get("work"):
        parts.append(plural(v["work"], "AI session"))
    if v.get("notes"):
        parts.append(plural(v["notes"], "note"))
    if v.get("digital"):
        parts.append(f"{v['digital']} digital")
    return " · ".join(parts)


def month_view(y: int, m: int, providers) -> dict:
    first = date(y, m, 1)
    last = date(y + (m == 12), m % 12 + 1, 1) - timedelta(days=1)
    raw, cov = _collect(providers, first, last)
    days = [{"day": d.isoformat(), "dom": d.day, "weekday": d.isoweekday(),
             "values": _values(d.isoformat(), raw.get(d.isoformat(), {}), cov)} for d in _span(first, last)]
    totals = {k: sum(x["values"][k] or 0 for x in days) for k in KEYS}
    return {"month": f"{y:04d}-{m:02d}", "label": f"{MONTHS[m - 1]} {y}", "first_weekday": first.isoweekday(),
            "days": days, "totals": totals, "metrics": metrics_meta()}


def year_view(y: int, providers) -> dict:
    first, last = date(y, 1, 1), date(y, 12, 31)
    raw, cov = _collect(providers, first, last)
    days = {d.isoformat(): _values(d.isoformat(), raw.get(d.isoformat(), {}), cov) for d in _span(first, last)}
    months = []
    for m in range(1, 13):
        vals = [v for ds, v in days.items() if ds.startswith(f"{y:04d}-{m:02d}")]
        row = {"month": f"{y:04d}-{m:02d}", "label": MONTHS[m - 1][:3]}
        for k in KEYS:
            row[k] = sum(v[k] or 0 for v in vals) if any(v[k] is not None for v in vals) else None
        months.append(row)
    return {"year": y, "first_weekday": first.isoweekday(), "days": days, "months": months, "metrics": metrics_meta()}


def on_this_day(d: date, providers, max_years: int = 20) -> list[dict]:
    froms = [s["from"] for s in (safe_status(p) for p in providers) if s.get("state") == "ok" and s.get("from")]
    earliest = int(min(froms)[:4]) if froms else d.year
    out = []
    for k in range(1, max_years + 1):
        y = d.year - k
        if y < earliest:
            break
        try:
            t = d.replace(year=y)
        except ValueError:  # 29/02
            t = d.replace(year=y, day=28)
        raw, _ = _collect(providers, t - timedelta(days=3), t + timedelta(days=3))
        for off in (0, -1, 1, -2, 2, -3, 3):
            ds = (t + timedelta(days=off)).isoformat()
            vals = raw.get(ds) or {}
            if any(vals.values()):
                out.append({"day": ds, "years_ago": k, "offset": off, "values": vals, "summary": counts_summary(vals)})
                break
    return out


def nearest_days(d: date, providers, window: int = 120) -> dict:
    raw, _ = _collect(providers, d - timedelta(days=window), d + timedelta(days=window))
    ds = d.isoformat()
    filled = sorted(k for k, v in raw.items() if any(v.values()))
    prev = [k for k in filled if k < ds]
    nxt = [k for k in filled if k > ds]
    return {"prev": prev[-1] if prev else None, "next": nxt[0] if nxt else None}


def search_all(q: str, providers, kinds: Optional[set] = None, frm: Optional[str] = None, to: Optional[str] = None,
               limit: int = 300) -> dict:
    hits: list = []
    for p in providers:
        if safe_status(p)["state"] != "ok":
            continue
        try:
            hits += p.search(q, limit)
        except Exception:
            continue
    if frm:
        hits = [h for h in hits if (h["day"] or "") >= frm]
    if to:
        hits = [h for h in hits if (h["day"] or "") <= to]
    kinds_count = dict(Counter(h["kind"] for h in hits))
    if kinds:
        hits = [h for h in hits if h["kind"] in kinds]
    hits.sort(key=lambda h: (h["day"] or "", h["ts"] or ""), reverse=True)
    truncated = len(hits) > limit
    hits = hits[:limit]
    days: list = []
    for h in hits:
        if not days or days[-1]["day"] != h["day"]:
            days.append({"day": h["day"], "items": []})
        days[-1]["items"].append(h)
    return {"q": q, "total": len(hits), "truncated": truncated, "kinds": kinds_count, "days": days}
```

```python
"""Fork-local (ver LOCAL_CHANGES.md §11): API do painel "Life" — linha do tempo unificada.

Spec: docs/superpowers/specs/2026-11-06-life-timeline-design.md. Lê, SOMENTE LEITURA, o lifelog.sqlite do NAS (Takeout,
bind /data/lifelog:/app/lifelog:ro), as Finanças (data/finance) e o Uso de IA (/app/ai-usage), pelos provedores de
src/life. Todas as rotas exigem admin.

  GET /api/lifelog/meta        hoje (UTC-3), intervalo coberto e estado das fontes
  GET /api/lifelog/day         página do dia montada no servidor (src/life/assemble.py) + dias vizinhos se vazio
  GET /api/lifelog/month       valores por dia de um mês (YYYY-MM)
  GET /api/lifelog/year        valores por dia/mês de um ano, métricas e faixas fixas
  GET /api/lifelog/onthisday   o mesmo dia em anos anteriores (±3 dias)
  GET /api/lifelog/search      busca em todas as fontes, agrupada por dia
  GET /api/lifelog/sources     estado de cada fonte
"""
from __future__ import annotations

import re
from datetime import date
from typing import Optional

from fastapi import APIRouter, Depends, HTTPException, Query

from core.middleware import require_admin
from src.life import registry
from src.life.assemble import assemble_day
from src.life.common import safe_status, today
from src.life.views import month_view, nearest_days, on_this_day, search_all, year_view

DAY_RE = re.compile(r"^\d{4}-\d{2}-\d{2}$")
MONTH_RE = re.compile(r"^(\d{4})-(\d{2})$")


def _date(v: Optional[str], name: str) -> Optional[date]:
    if not v:
        return None
    try:
        if not DAY_RE.match(v):
            raise ValueError(v)
        return date.fromisoformat(v)
    except ValueError:
        raise HTTPException(400, f"invalid {name} (YYYY-MM-DD)")


def setup_lifelog_routes() -> APIRouter:
    r = APIRouter(prefix="/api/lifelog", tags=["lifelog"], dependencies=[Depends(require_admin)])

    @r.get("/meta")
    def meta():
        sources = [safe_status(p) for p in registry.providers()]
        froms = [s["from"] for s in sources if s.get("from")]
        tos = [s["to"] for s in sources if s.get("to")]
        return {"today": today().isoformat(), "sources": sources,
                "range": {"from": min(froms) if froms else None, "to": max(tos) if tos else None}}

    @r.get("/day")
    def day(d: str):
        dd = _date(d, "d")
        if dd is None:
            raise HTTPException(400, "d is required")
        ps = registry.providers()
        out = assemble_day(dd, ps)
        out["nearest"] = None if (out["blocks"] or out["all_day"]) else nearest_days(dd, ps)
        return out

    @r.get("/month")
    def month(m: str):
        mm = MONTH_RE.match(m or "")
        if not mm or not 1 <= int(mm.group(2)) <= 12:
            raise HTTPException(400, "invalid m (YYYY-MM)")
        return month_view(int(mm.group(1)), int(mm.group(2)), registry.providers())

    @r.get("/year")
    def year(y: str):
        if not re.fullmatch(r"\d{4}", y or ""):
            raise HTTPException(400, "invalid y (YYYY)")
        return year_view(int(y), registry.providers())

    @r.get("/onthisday")
    def onthisday(d: str):
        dd = _date(d, "d")
        if dd is None:
            raise HTTPException(400, "d is required")
        return on_this_day(dd, registry.providers())

    @r.get("/search")
    def search(q: str = "", kinds: str = "", frm: Optional[str] = Query(None, alias="from"),
               to: Optional[str] = None, limit: int = 300):
        q = q.strip()
        if len(q) < 2:
            raise HTTPException(400, "query too short (min 2 characters)")
        a, b = _date(frm, "from"), _date(to, "to")
        ks = {k for k in kinds.split(",") if k} or None
        return search_all(q, registry.providers(), kinds=ks, frm=a.isoformat() if a else None,
                          to=b.isoformat() if b else None, limit=max(1, min(limit, 1000)))

    @r.get("/sources")
    def sources():
        return [safe_status(p) for p in registry.providers()]

    return r
```

```js
// Sem DOM: testado por tests/life_format.test.mjs. Datas são strings ISO (AAAA-MM-DD) tratadas em UTC.
const WD = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'];
const WD_LONG = ['Sunday', 'Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday'];
const MON = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
const MON_LONG = ['January', 'February', 'March', 'April', 'May', 'June', 'July', 'August', 'September', 'October',
  'November', 'December'];
const SYMBOLS = { BRL: 'R$', USD: 'US$', EUR: '€', GBP: '£' };
const DAY_RE = /^\d{4}-\d{2}-\d{2}$/;
const MONTH_RE = /^\d{4}-\d{2}$/;
const YEAR_RE = /^\d{4}$/;

export function parseIso(d) {
  const [y, m, dd] = d.split('-').map(Number);
  return new Date(Date.UTC(y, m - 1, dd));
}
const iso = dt => dt.toISOString().slice(0, 10);

export function shiftDay(d, n) {
  const x = parseIso(d);
  x.setUTCDate(x.getUTCDate() + n);
  return iso(x);
}
export function shiftMonth(m, n) {
  const [y, mm] = m.split('-').map(Number);
  const t = y * 12 + (mm - 1) + n;
  return `${Math.floor(t / 12)}-${String((t % 12) + 1).padStart(2, '0')}`;
}
export function fmtDate(d) {
  const x = parseIso(d);
  return `${WD[x.getUTCDay()]}, ${x.getUTCDate()} ${MON[x.getUTCMonth()]} ${x.getUTCFullYear()}`;
}
export function fmtDateLong(d) {
  const x = parseIso(d);
  return `${WD_LONG[x.getUTCDay()]}, ${x.getUTCDate()} ${MON_LONG[x.getUTCMonth()]} ${x.getUTCFullYear()}`;
}
export function fmtMonth(m) {
  const [y, mm] = m.split('-').map(Number);
  return `${MON_LONG[mm - 1]} ${y}`;
}
export const monthName = mm => MON_LONG[mm - 1];
export const yearsAgo = n => (n === 1 ? '1 year ago' : `${n} years ago`);

export function fmtMoney(cents, cur = 'BRL', { decimals } = {}) {
  const v = Math.abs(cents) / 100;
  const d = decimals ?? (v >= 100 ? 0 : 2);
  const sym = SYMBOLS[cur] || cur;
  return `${cents < 0 ? '−' : ''}${sym} ${v.toLocaleString('en-US', { minimumFractionDigits: d, maximumFractionDigits: d })}`;
}
export function compact(n) {
  if (n == null) return '—';
  const a = Math.abs(n);
  for (const [div, suf] of [[1e6, 'M'], [1e3, 'k']]) {
    if (a >= div) return `${(n / div).toFixed(1).replace(/\.0$/, '')}${suf}`;
  }
  return String(Math.round(n));
}
export function escapeHtml(s) {
  return String(s ?? '').replace(/[&<>"']/g, c => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]));
}
export function highlight(text, q) {
  const t = String(text ?? '');
  const needle = String(q ?? '').trim().toLowerCase();
  if (!needle) return escapeHtml(t);
  const lower = t.toLowerCase();
  let out = '';
  let i = 0;
  for (;;) {
    const j = lower.indexOf(needle, i);
    if (j < 0) { out += escapeHtml(t.slice(i)); break; }
    out += `${escapeHtml(t.slice(i, j))}<mark>${escapeHtml(t.slice(j, j + needle.length))}</mark>`;
    i = j + needle.length;
  }
  return out;
}
export function parseHash(h) {
  const s = String(h || '').replace(/^#/, '');
  if (s !== 'life' && !s.startsWith('life/')) return null;
  const rest = s.slice(5);
  if (!rest) return { view: null };
  if (rest === 'search' || rest.startsWith('search?')) {
    return { view: 'search', q: new URLSearchParams(rest.split('?')[1] || '').get('q') || '' };
  }
  if (DAY_RE.test(rest)) return { view: 'day', day: rest };
  if (MONTH_RE.test(rest)) return { view: 'month', month: rest };
  if (YEAR_RE.test(rest)) return { view: 'year', year: rest };
  return { view: null };
}
export function buildHash(st) {
  if (st.view === 'search') return `#life/search?q=${encodeURIComponent(st.q || '')}`;
  if (st.view === 'month') return `#life/${st.day.slice(0, 7)}`;
  if (st.view === 'year') return `#life/${st.day.slice(0, 4)}`;
  return `#life/${st.day}`;
}
```

```js
// static/js/life/api.js — cliente das rotas /api/lifelog/* (cancela a requisição anterior com a mesma chave).
const API = '/api/lifelog';
const inflight = new Map();

export async function api(path, params = {}, { key = path } = {}) {
  inflight.get(key)?.abort();
  const ctl = new AbortController();
  inflight.set(key, ctl);
  const qs = new URLSearchParams(Object.entries(params).filter(([, v]) => v !== '' && v != null));
  try {
    const r = await fetch(`${API}/${path}${qs.toString() ? `?${qs}` : ''}`, { credentials: 'same-origin', signal: ctl.signal });
    if (!r.ok) {
      const body = await r.json().catch(() => ({}));
      throw new Error(body.detail || `HTTP ${r.status}`);
    }
    return await r.json();
  } finally {
    if (inflight.get(key) === ctl) inflight.delete(key);
  }
}

export const isAbort = e => e?.name === 'AbortError';
```

```js
import { compact, escapeHtml, fmtMoney } from './format.js';

export function levelOf(v, bands) {
  if (v == null) return 'none';
  let lvl = 0;
  bands.forEach((b, i) => { if (v >= b) lvl = i + 1; });
  return lvl;

export function fmtMetric(key, v, short = false) {
  if (v == null) return 'no data';
  if (key === 'spent') return fmtMoney(v, 'BRL', { decimals: 0 });
  if (short) return compact(v);

}

}

export function legend(metric) {
  const b = metric.bands;
  const f = x => fmtMetric(metric.key, x, true);
  const labels = b.map((x, i) => (i === 0 ? `< ${f(b[1])}` : i === b.length - 1 ? `${f(x)}+` : `${f(x)}–${f(b[i + 1])}`));

}
```

```js
function restore() {
  try {
    const s = JSON.parse(localStorage.getItem(LS_KEY) || '{}');
    if (VIEWS[s.view]) st.view = s.view;
    if (/^\d{4}-\d{2}-\d{2}$/.test(s.day || '')) st.day = s.day;

function persist() {

}
const isOpen = () => !!els && !els.panel.classList.contains('hidden');
const todayIso = () => st.meta?.today || st.day || new Date().toISOString().slice(0, 10);

function raise() {

    els.panel.style.setProperty('z-index', String(nextToolWindowZ({ exclude: els.panel, current: getComputedStyle(els.panel).zIndex, floor: 300 })), 'important');

function buildPanel() {

function crumbs() {

function go(patch = {}, { replace = false } = {}) {
  Object.assign(st, patch);
  if (!st.day) st.day = todayIso();
  const h = buildHash(st);
  if (location.hash !== h) {
    try { history[replace ? 'replaceState' : 'pushState'](null, '', h); } catch (_) { location.hash = h; }
  }
  persist();
  render();
}

function step(n) {
  if (st.view === 'day') go({ day: shiftDay(st.day, n) });
  else if (st.view === 'month') go({ day: `${shiftMonth(st.day.slice(0, 7), n)}-01` });
  else if (st.view === 'year') go({ day: `${Number(st.day.slice(0, 4)) + n}-01-01` });
}
function zoomOut() {
  if (st.view === 'day') go({ view: 'month' });
  else if (st.view === 'month') go({ view: 'year' });
}
function goToday() { go({ view: st.view === 'search' ? 'day' : st.view, day: todayIso() }); }

function onKey(e) {
  const typing = e.target.closest?.('input, textarea, select, [contenteditable="true"]');
  if (e.key === '/' && !typing) { e.preventDefault(); els.input.focus(); els.input.select(); return; }
  if (typing || e.altKey || e.ctrlKey || e.metaKey || e.shiftKey) return;
  if (e.key === 'ArrowLeft' && st.view !== 'search') { e.preventDefault(); step(-1); }
  else if (e.key === 'ArrowRight' && st.view !== 'search') { e.preventDefault(); step(1); }
  else if (e.key === 'ArrowUp' && els.body.scrollTop === 0 && (st.view === 'day' || st.view === 'month')) { e.preventDefault(); zoomOut(); }
  else if (e.key === 't') goToday();
}

async function render() {
  const mine = ++seq;
  lastHash = buildHash(st);
  crumbs();
  els.panel.dataset.view = st.view;
  els.nav.hidden = st.view === 'search';
  els.input.value = st.view === 'search' ? st.q : '';
  els.body.setAttribute('aria-busy', 'true');
  els.body.innerHTML = skeleton(st.view);
  els.body.scrollTop = 0;
  const ctx = { st, go, api, isCurrent: () => mine === seq };
  try {
    await VIEWS[st.view](els.body, ctx);
  } catch (e) {
    if (isAbort(e) || mine !== seq) return;
    els.body.innerHTML = emptyState('other', 'Could not load this view', escapeHtml(e.message));
  } finally {
    if (mine === seq) els.body.removeAttribute('aria-busy');
  }
}

async function openPanel({ view, day, today = false, replace = false } = {}) {
  buildPanel();
  els.panel.classList.remove('hidden');
  raise();
  if (!st.meta) {
    try {
      st.meta = await api('meta');
    } catch (e) {
      if (isAbort(e)) return;
      els.body.innerHTML = emptyState('life', 'Life is not available', escapeHtml(e.message));
      return;
    }
  }
  if (view) st.view = view;
  if (day) st.day = day;
  if (today || !st.day) st.day = st.meta.today;
  go({}, { replace });
  els.body.focus({ preventScroll: true });
}

function fromHash() {
  const h = parseHash(location.hash);
  if (!h) return;
  if (location.hash === lastHash && isOpen()) return;
  if (h.view === 'day') Object.assign(st, { view: 'day', day: h.day });
  else if (h.view === 'month') Object.assign(st, { view: 'month', day: st.day.startsWith(h.month) ? st.day : `${h.month}-01` });
  else if (h.view === 'year') Object.assign(st, { view: 'year', day: st.day.startsWith(h.year) ? st.day : `${h.year}-01-01` });
  else if (h.view === 'search') Object.assign(st, { view: 'search', q: h.q });
  if (isOpen() && st.meta) { persist(); render(); } else openPanel({ replace: true });
}

function mountSidebar() {

    item.innerHTML = `${icon(name, 14)}<span class="grow">${text}</span>`;
    item.addEventListener('click', fn);
    list.append(item);
  };
  add('calendar', 'Today', () => openPanel({ view: 'day', today: true }));
  add('search', 'Search everything', async () => { await openPanel({ view: 'search' }); els?.input.focus(); });
}

restore();
```

```js
// static/js/life/day.js — visão do dia do Life: cabeçalho com resumo, chips de filtro, faixa All day, espinha de
// blocos (períodos na fase 1) com grupos "Digital" recolhidos, e coluna com números, On this day e fontes.
// O servidor monta tudo (src/life/assemble.py); aqui só se desenha e se liga o filtro/expansões.
import { icon } from './icons.js';
import { escapeHtml as esc, fmtDate, fmtDateLong, fmtMoney, yearsAgo } from './format.js';
import { KIND_LABEL, KIND_ORDER, emptyState, openLink, sourcesFooter } from './parts.js';

const PERIOD_ICON = { night: 'night', morning: 'morning', afternoon: 'afternoon', evening: 'evening' };
const CAP = 5;

export async function renderDay(body, ctx) {
  const { st } = ctx;
  const data = await ctx.api('day', { d: st.day }, { key: 'view' });
  if (!ctx.isCurrent()) return;
  if (st.filter !== 'all' && !data.counts_by_kind[st.filter]) st.filter = 'all';
  const empty = !data.blocks.length && !data.all_day.length;
  body.innerHTML = `
    <article class="life-day">
      <header class="life-dayhead">
        <h2>${esc(fmtDateLong(data.day))}</h2>
        ${data.summary.parts.length ? `<p class="life-summary">${data.summary.parts.map(p => `<span class="life-sumpart k-${esc(p.kind)}">${icon(p.kind, 13)}${esc(p.text)}</span>`).join('')}</p>` : ''}
      </header>
      <div class="life-cols">
        <section class="life-main" aria-label="Timeline">
          ${empty ? emptyDay(data) : `${chips(data.counts_by_kind, st.filter)}${allDay(data.all_day)}<ol class="life-spine">${data.blocks.map(block).join('')}</ol>`}
        </section>
        <aside class="life-side" aria-label="Day details">
          ${numbers(data.stats)}
          <section class="life-card life-otd" aria-label="On this day"><h3>${icon('calendar', 14)}On this day</h3><div class="life-otd-body"><div class="life-skel" style="width:70%"></div><div class="life-skel" style="width:50%"></div></div></section>
          ${sourcesFooter(data.sources)}
        </aside>
      </div>
    </article>`;
  const root = body.querySelector('.life-day');
  root.addEventListener('click', e => {
    const chip = e.target.closest('.life-chip[data-filter]');
    if (chip) { st.filter = chip.dataset.filter; applyFilter(root, st.filter); return; }
    const more = e.target.closest('[data-more]');
    if (more) { more.closest('.life-dsrc').classList.add('is-expanded'); more.remove(); return; }
    const goDay = e.target.closest('[data-go-day]');
    if (goDay) ctx.go({ view: 'day', day: goDay.dataset.goDay });
  });
  applyFilter(root, st.filter);
  loadOnThisDay(root, ctx, data.day);
}

function chips(counts, filter) {
  const kinds = KIND_ORDER.filter(k => counts[k]);
  if (kinds.length < 2) return '';
  const total = kinds.reduce((a, k) => a + counts[k], 0);
  const btn = (k, label, n) => `<button type="button" class="life-chip" data-filter="${k}" aria-pressed="${filter === k}">${k === 'all' ? '' : icon(k, 13)}${esc(label)}<span class="life-chip-n">${n}</span></button>`;
  return `<div class="life-chips" role="toolbar" aria-label="Filter by kind">${btn('all', 'All', total)}${kinds.map(k => btn(k, KIND_LABEL[k], counts[k])).join('')}</div>`;
}

function allDay(items) {
  if (!items.length) return '';
  return `<section class="life-allday" aria-label="All day"><h3 class="life-eyebrow">All day</h3><ul class="life-entries">${items.map(entry).join('')}</ul></section>`;
}

function block(b) {
  return `<li class="life-block" data-key="${esc(b.key || b.kind)}">
    <div class="life-block-h">${icon(PERIOD_ICON[b.key] || 'place', 15)}<span class="life-block-t">${esc(b.label)}</span><span class="life-block-r">${esc(b.start)}–${esc(b.end)}</span></div>
    <ul class="life-entries">${b.entries.map(e => (e.kind === 'digital_group' ? digital(e) : entry(e))).join('')}</ul>
    <p class="life-block-none">Nothing of this kind here.</p>
  </li>`;
}

function timeCell(start, end) {
  if (!start) return '<time></time>';
  return `<time>${esc(start)}${end && end !== start ? `<span class="life-time-end">–${esc(end)}</span>` : ''}</time>`;
}

function entry(it) {
  const m = it.meta || {};
  let amount = '';
  if (it.kind === 'money') {
    const cls = `life-amount${m.amount_cents > 0 ? ' is-in' : ''}${m.neutral ? ' is-neutral' : ''}`;
    const tip = m.neutral ? ' title="Between your own accounts — not counted as spending or income"' : '';
    amount = `<span class="${cls}"${tip}>${esc(fmtMoney(m.amount_cents, m.currency, { decimals: 2 }))}</span>`;
  }
  return `<li class="life-entry k-${esc(it.kind)}" data-kind="${esc(it.kind)}">
    ${timeCell(it.time, it.end_time)}<span class="life-ic">${icon(it.kind, 15)}</span>
    <div class="life-entry-b"><div class="life-entry-t">${esc(it.title)}</div>${it.subtitle ? `<div class="life-entry-s">${esc(it.subtitle)}</div>` : ''}</div>
    ${amount}${openLink(it.open_url)}
  </li>`;
}

function digital(g) {
  const sources = g.sources.map(s => {
    const rows = s.items.map((r, i) => `<li class="life-drow${i >= CAP ? ' is-more' : ''}${r.unavailable ? ' is-unavailable' : ''}">
        <time>${esc(r.times[0])}</time>
        <div class="life-entry-b"><div class="life-entry-t">${esc(r.title)}${r.count > 1 ? `<span class="life-times" title="${esc(r.times.join(', '))}">×${r.count}</span>` : ''}</div>${r.subtitle ? `<div class="life-entry-s">${esc(r.subtitle)}</div>` : ''}</div>
        ${openLink(r.open_url)}</li>`).join('');
    const more = s.items.length > CAP ? `<button type="button" class="life-more" data-more>Show ${s.items.length - CAP} more</button>` : '';
    const views = s.views !== s.count ? ` · ${s.views} views` : '';
    return `<section class="life-dsrc"><h4>${icon(s.group, 13)}${esc(s.label)}<span>${s.count}${views}</span></h4><ul>${rows}</ul>${more}</section>`;
  }).join('');
  return `<li class="life-entry k-digital" data-kind="digital">
    <details class="life-dg">
      <summary>${timeCell(g.time, g.end_time)}<span class="life-ic">${icon('digital', 15)}</span>
        <div class="life-entry-b"><div class="life-entry-t"><b>Digital</b> · ${esc(g.summary)}</div></div>${icon('chevron-down', 14)}</summary>
      <div class="life-dg-body">${sources}</div>
    </details></li>`;
}

function numbers(stats) {
  if (!stats.length) return '';
  return `<section class="life-card" aria-label="Day in numbers"><h3>${icon('health', 14)}Day in numbers</h3>
    <dl class="life-kpis">${stats.map(s => `<div class="life-kpi"><dt>${esc(s.label)}</dt><dd>${esc(s.value)}</dd></div>`).join('')}</dl></section>`;
}

function emptyDay(data) {
  const n = data.nearest || {};
  const btn = (d, before) => (d
    ? `<button type="button" class="life-btn" data-go-day="${esc(d)}">${before ? '← ' : ''}${esc(fmtDate(d))}${before ? '' : ' →'}</button>`
    : '');
  return emptyState('calendar', 'Nothing recorded on this day', 'None of the connected sources has anything for this date.',
    btn(n.prev, true) + btn(n.next, false));
}

function applyFilter(root, filter) {
  root.querySelectorAll('.life-chip[data-filter]').forEach(c => c.setAttribute('aria-pressed', String(c.dataset.filter === filter)));
  root.querySelectorAll('.life-chip[data-filter]').forEach(c => c.classList.toggle('is-on', c.dataset.filter === filter));
  root.querySelectorAll('.life-entry').forEach(en => { en.hidden = filter !== 'all' && en.dataset.kind !== filter; });
  root.querySelectorAll('.life-block, .life-allday').forEach(b => {
    b.classList.toggle('is-filtered-empty', filter !== 'all' && !b.querySelector(`.life-entry[data-kind="${filter}"]`));
  });
}

async function loadOnThisDay(root, ctx, day) {
  const box = root.querySelector('.life-otd-body');
  try {
    const rows = await ctx.api('onthisday', { d: day }, { key: 'otd' });
    if (!ctx.isCurrent()) return;
    box.innerHTML = rows.length
      ? `<ul class="life-otd-list">${rows.map(r => `<li><button type="button" data-go-day="${esc(r.day)}">
          <span class="life-otd-when">${esc(yearsAgo(r.years_ago))}${r.offset ? ` <em>(${r.offset > 0 ? '+' : '−'}${Math.abs(r.offset)} day${Math.abs(r.offset) === 1 ? '' : 's'})</em>` : ''}</span>
          <span class="life-otd-date">${esc(fmtDate(r.day))}</span><span class="life-otd-sum">${esc(r.summary)}</span></button></li>`).join('')}</ul>`
      : '<p class="life-muted">Nothing from this date in earlier years.</p>';
  } catch (e) {
    if (ctx.isCurrent()) box.innerHTML = '<p class="life-muted">Could not load earlier years.</p>';
  }
}
```

```js
// static/js/life/year.js — ano: mapa de calor por dia (semanas em colunas, linhas seg..dom), legenda com faixas fixas
// e barras por mês. Clique no dia → dia; no mês → mês. Substitui o antigo "Trends".
import { escapeHtml as esc, monthName } from './format.js';
import { fmtMetric, legend, levelOf, metricBar, wireMetric } from './metrics.js';

export async function renderYear(body, ctx) {
  const { st } = ctx;
  const y = st.day.slice(0, 4);
  const data = await ctx.api('year', { y }, { key: 'view' });
  if (!ctx.isCurrent()) return;
  const metric = data.metrics.find(m => m.key === st.metric) || data.metrics[0];
  const days = Object.keys(data.days).sort();
  const lead = data.first_weekday - 1;
  const today = st.meta?.today;
  const pads = Array.from({ length: lead }, () => '<span class="life-hcell is-pad" aria-hidden="true"></span>');
  const cells = days.map(d => {
    const v = data.days[d][metric.key];
    const label = `${d} · ${fmtMetric(metric.key, v)}`;

    return `<button type="button" class="life-hmonth" style="grid-column:${col}" data-go-month="${m.month}">${esc(m.label)}</button>`;

  const max = Math.max(1, ...data.months.map(m => m[metric.key] || 0));
  const bars = data.months.map(m => {
    const v = m[metric.key];
    const h = v ? Math.max(4, Math.round((v / max) * 100)) : 0;
    return `<button type="button" class="life-bar" data-go-month="${m.month}" title="${esc(monthName(Number(m.month.slice(5))))}: ${esc(fmtMetric(metric.key, v))}">

    <header class="life-viewhead"><h2>${esc(y)}</h2>${metricBar(data.metrics, metric.key)}</header>
    <div class="life-heat" aria-label="${esc(metric.label)} per day">
      <div class="life-hmonths">${monthLabels}</div>
      <div class="life-hgrid">${pads.join('')}${cells.join('')}</div>
    </div>
    ${legend(metric)}
    <section class="life-card"><h3>${esc(metric.label)} per month</h3><div class="life-bars">${bars}</div></section>

    const m = e.target.closest('[data-go-month]');

  });
}
```

```js
// Fork-local: checagem visual do painel Life no Chromium headless DENTRO do container do Odysseus.
// Uso (no host):
//   docker cp scripts/life_ui_check.cjs odysseus-odysseus-1:/tmp/ && docker exec odysseus-odysseus-1 node /tmp/life_ui_check.cjs 2023-03-14
// Entra pelo proxy do NAS com o segredo do PRÓPRIO ambiente do container (nunca sai dele), percorre dia/mês/ano/busca
// em 1440 px (escuro e claro) e 390 px, grava /tmp/life-*.png e sai com código 1 se houver erro de JS ou 4xx/5xx.
// As capturas têm dados pessoais: apagar depois de revisar.
const fs = require('fs');
const path = require('path');

const base = '/app/.npm/_npx';
const pw = fs.readdirSync(base).map(d => path.join(base, d, 'node_modules/playwright-core')).find(p => fs.existsSync(p));
const { chromium } = require(pw);
const LIGHT = { '--bg': '#f3f5f7', '--fg': '#26313a', '--panel': '#ffffff', '--border': '#cbd3da' };

(async () => {
  const day = process.argv[2] || '2023-03-14';
  const remote = (process.env.TRUSTED_PROXY_USER_MAP || '').split(',')[0].split(':')[0].trim();
  const headers = { 'X-Odysseus-Proxy-Secret': process.env.TRUSTED_PROXY_SECRET || '', 'Remote-User': remote };
  const browser = await chromium.launch({ executablePath: '/usr/bin/chromium', args: ['--no-sandbox'] });
  const errors = [];
  const shots = [];
  for (const [w, h, theme] of [[1440, 900, 'dark'], [1440, 900, 'light'], [390, 844, 'dark']]) {
    const ctx = await browser.newContext({ viewport: { width: w, height: h }, serviceWorkers: 'block', extraHTTPHeaders: headers });
    const page = await ctx.newPage();
    const tag = `${w}-${theme}`;
    page.on('pageerror', e => errors.push(`${tag} pageerror: ${e.message}`));
    page.on('console', m => { if (m.type() === 'error') errors.push(`${tag} console: ${m.text()}`); });
    page.on('response', r => { if (r.status() >= 400 && /\/static\/|\/api\/lifelog/.test(r.url())) errors.push(`${tag} ${r.status()} ${r.url()}`); });
    await page.goto(`http://127.0.0.1:7000/#life/${day}`, { waitUntil: 'networkidle', timeout: 60000 });
    if (theme === 'light') await page.evaluate(v => { for (const [k, x] of Object.entries(v)) document.documentElement.style.setProperty(k, x); }, LIGHT);
    const snap = async name => { const f = `/tmp/life-${name}-${tag}.png`; await page.screenshot({ path: f }); shots.push(f); };
    await page.waitForSelector('#life-panel .life-day, #life-panel .life-empty', { timeout: 20000 });
    await page.waitForTimeout(1200);
    await snap('day');
    await page.evaluate(() => document.querySelector('#life-panel .life-dg > summary')?.click());
    await page.waitForTimeout(300);
    await snap('day-open');
    for (const [view, hash] of [['month', day.slice(0, 7)], ['year', day.slice(0, 4)]]) {
      await page.evaluate(x => { location.hash = `#life/${x}`; }, hash);
      await page.waitForSelector(`#life-panel .life-${view}`, { timeout: 20000 });
      await page.waitForTimeout(800);
      await snap(view);
    }
    await page.evaluate(() => { location.hash = '#life/search?q=youtube'; });
    await page.waitForSelector('#life-panel .life-search-view, #life-panel .life-empty', { timeout: 20000 });
    await page.waitForTimeout(800);
    await snap('search');
    await ctx.close();
  }
  await browser.close();
  console.log(JSON.stringify({ errors, shots }, null, 1));
  process.exit(errors.length ? 1 : 0);
})().catch(e => { console.error('FALHOU', e); process.exit(2); });
```
